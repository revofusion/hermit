14/// [`detcore_model::config::config_wire_fingerprint`], extended with the
15/// revision of Detcore's global RPC (`crate::tool_global::GLOBAL_RPC_REVISION`).
16///
17/// A runtime that embeds Detcore (SaBRe's plugin, the LiteInst runtime that
18/// in-guest-trap also loads, DBT's) sends `GlobalRequest`s to its coordinator
19/// as positional bincode and decodes the `GlobalResponse`s. Both enums live
20/// in this crate, outside the sources the configuration's fingerprint hashes,
21/// so a variant one build had and the other lacked went unnoticed: a newer
22/// runtime's request failed to decode in the middle of a run, and an older
23/// runtime never sent it (and so, for one, kept the host's inotify rename
24/// cookies). Every such runtime compares this fingerprint with its
25/// coordinator's before it decodes anything, so the pair is refused by name.
26/// The revision stands for the enums' source, which, unlike the
27/// configuration's, is not embedded: these runtimes are loaded into the guest,
28/// where every byte they grow by moves the guest's own mappings.
29pub fn config_wire_fingerprint() -> String {
30    fingerprint_with_revision(
31        &detcore_model::config::config_wire_fingerprint(),
32        crate::tool_global::GLOBAL_RPC_REVISION,
33    )
34}
35
36/// What differs when a coordinator announced `expected` as its
37/// [`config_wire_fingerprint`] and this build's is another, or `None` when
38/// they are equal. The runtimes report this before they decode the
39/// coordinator's configuration (<https://github.com/rrnewton/hermit/issues/3986>).
40pub fn config_fingerprint_mismatch(expected: &std::ffi::OsStr) -> Option<String> {
41    let ours = config_wire_fingerprint();
42    (expected.to_str() != Some(ours.as_str())).then(|| {
43        format!(
44            "its configuration and clock RPC fingerprint is {ours}, Hermit's is {}",
45            expected.to_string_lossy()
46        )
47    })
48}

50/// FNV-1a over `bytes`, each part length-prefixed in a domain of its own,
51/// as `detcore_model`'s fingerprint hashes its material. A mismatch
52/// detector, not a security boundary.
53fn fnv_domains(parts: &[(u8, &[u8])]) -> String {
54    let mut hash: u64 = 0xcbf2_9ce4_8422_2325;
55    for (domain, bytes) in parts {
56        for byte in std::iter::once(domain)
57            .chain((bytes.len() as u64).to_le_bytes().iter())
58            .chain(bytes.iter())
59        {
60            hash ^= u64::from(*byte);
61            hash = hash.wrapping_mul(0x1000_0000_01b3);
62        }
63    }
64    format!("{hash:016x}")
65}
66
67/// The fingerprint of configuration fingerprint `base` at global RPC
68/// revision `revision`.
69fn fingerprint_with_revision(base: &str, revision: u32) -> String {
70    fnv_domains(&[(0, base.as_bytes()), (1, &revision.to_le_bytes())])
71}

77    /// Every revision of the global RPC, oldest first, with the digest of the
78    /// code of its two enums and its `Resources` payload. A change to any of
79    /// these appends a row and sets
80    /// `GLOBAL_RPC_REVISION` to its revision.
81    const GLOBAL_RPC_REVISIONS: &[(u32, &str, &str)] = &[
82        // GlobalRequest::VirtualInotifyCookies and its response, the first
83        // revision the fingerprint carries.
84        (1, "a61c88ecb1abcb5e", "9b7e11bf5f08e6ae"),
85        // GlobalRequest::RegisterDescriptorDirectory and DescriptorDirectory and
86        // their responses (package K slice F1).
87        (2, "caee2c37469869d4", "9b7e11bf5f08e6ae"),
88        // GlobalRequest::RegisterPlaceholder, Placeholder, PlaceholderFlock,
89        // PlaceholderWriteHint and PlaceholderCopies and their responses, in
90        // place of F1's two (package K slice P1).
91        (3, "b3d7410a78b56ae0", "9b7e11bf5f08e6ae"),
92        // Resources::priority_row extends the positional request payload.
93        (4, "b3d7410a78b56ae0", "4e10b2de93d47d64"),
94    ];

96    /// The lines of code, without comments or blank lines, of the type that
97    /// `source` declares on the line `header`, through its closing brace.
98    fn definition(source: &str, header: &str) -> Vec<String> {
99        let mut lines = source.lines().skip_while(|line| *line != header);
100        assert_eq!(lines.next(), Some(header), "{header} is declared");
101        let mut code: Vec<String> = lines
102            .take_while(|line| *line != "}")
103            .map(str::trim)
104            .filter(|line| !line.is_empty() && !line.starts_with("//"))
105            .map(str::to_owned)
106            .collect();
107        code.insert(0, header.to_owned());
108        code
109    }

111    /// A digest of the code of `GlobalRequest` and `GlobalResponse` in
112    /// `source`: their variants, payload types, order and attributes, not
113    /// their documentation.
114    fn definition_digest(source: &str) -> String {
115        let request = definition(source, "pub enum GlobalRequest {").join("\n");
116        let response = definition(source, "pub enum GlobalResponse {").join("\n");
117        fnv_domains(&[(0, request.as_bytes()), (1, response.as_bytes())])
118    }
119
120    fn resources_definition_digest(source: &str) -> String {
121        let resources = definition(source, "pub struct Resources {").join("\n");
122        fnv_domains(&[(2, resources.as_bytes())])
123    }
124
125    /// The global RPC's revision names its enums and Resources as they are:
126    /// a variant or field added, removed, reordered or retyped changes the digest,
127    /// and the test then
128    /// fails until a new revision is recorded, which changes the fingerprint
129    /// every runtime compares with its coordinator's.
130    #[test]
131    fn the_global_rpc_revision_matches_its_definitions() {
132        let source = include_str!("tool_global.rs");
133        let digest = definition_digest(source);
134        let resources = include_str!("resources.rs");
135        let resources_digest = resources_definition_digest(resources);
136        let revision = crate::tool_global::GLOBAL_RPC_REVISION;
137        let next = GLOBAL_RPC_REVISIONS
138            .last()
139            .map_or(1, |(last, _, _)| last + 1);
140        assert_eq!(
141            GLOBAL_RPC_REVISIONS.last(),
142            Some(&(revision, digest.as_str(), resources_digest.as_str())),
143            "GlobalRequest, GlobalResponse or Resources changed: append ({next}, \
144             \"{digest}\", \"{resources_digest}\") to GLOBAL_RPC_REVISIONS and set \
145             GLOBAL_RPC_REVISION to {next}, so \
146             that runtimes and coordinators from either side of the change refuse each other"
147        );
148        for pair in GLOBAL_RPC_REVISIONS.windows(2) {
149            assert_eq!(pair[1].0, pair[0].0 + 1, "revisions count up by one");
150            assert_ne!(
151                (pair[1].1, pair[1].2),
152                (pair[0].1, pair[0].2),
153                "each revision changed the wire definitions"
154            );
155        }
156
157        // What the digest sees, and what it does not.
158        let documented = source.replacen(
159            "pub enum GlobalRequest {\n",
160            "pub enum GlobalRequest {\n    /// Documentation only.\n\n",
161            1,
162        );
163        assert_eq!(definition_digest(&documented), digest);
164        let added = source.replacen(
165            "pub enum GlobalRequest {\n",
166            "pub enum GlobalRequest {\n    Added(u32),\n",
167            1,
168        );
169        assert_ne!(definition_digest(&added), digest);
170        // Retype GlobalRequest's VirtualInotifyCookies wherever it sits: later
171        // revisions append variants after it.
172        let retyped = source.replacen(
173            "    VirtualInotifyCookies(Vec<u32>),\n",
174            "    VirtualInotifyCookies(Vec<u64>),\n",
175            1,
176        );
177        assert_ne!(retyped, source);
178        assert_ne!(definition_digest(&retyped), digest);
179
180        let documented = resources.replacen(
181            "pub struct Resources {\n",
182            "pub struct Resources {\n    /// Documentation only.\n\n",
183            1,
184        );
185        assert_ne!(documented, resources);
186        assert_eq!(resources_definition_digest(&documented), resources_digest);
187        let removed = resources.replacen("    pub(crate) priority_row: Option<u64>,\n", "", 1);
188        assert_ne!(removed, resources);
189        assert_ne!(resources_definition_digest(&removed), resources_digest);
190        let retyped =
191            resources.replacen("priority_row: Option<u64>", "priority_row: Option<u32>", 1);
192        assert_ne!(retyped, resources);
193        assert_ne!(resources_definition_digest(&retyped), resources_digest);
194    }

196    /// The published fingerprint is the configuration's extended with the
197    /// revision, so another revision is a mismatch, named with both values.
198    #[test]
199    fn the_fingerprint_carries_the_global_rpc_revision() {
200        let model = detcore_model::config::config_wire_fingerprint();
201        let ours = config_wire_fingerprint();
202        let revision = crate::tool_global::GLOBAL_RPC_REVISION;
203        assert_eq!(ours.len(), 16);
204        assert_eq!(ours, fingerprint_with_revision(&model, revision));
205        assert_ne!(ours, model);
206        let other = fingerprint_with_revision(&model, revision + 1);
207        assert_ne!(other, ours);
208        assert_eq!(
209            config_fingerprint_mismatch(std::ffi::OsStr::new(&ours)),
210            None
211        );
212        let mismatch = config_fingerprint_mismatch(std::ffi::OsStr::new(&other))
213            .expect("another revision is a mismatch");
214        assert!(
215            mismatch.contains(&ours) && mismatch.contains(&other),
216            "{mismatch}"
217        );
218    }
